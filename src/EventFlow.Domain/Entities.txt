
**Entities:


Tenant=> Type : ( Id,Name,Description )

User=> Type : ( Id,PhoneNumber,TenantId?,Name,Email,PasswordHash,Role?,IsActive )

Event=> Type : ( Id,TenantId,EventName,EventStatus,Description,Location,StartDate,EndDate,RegistrationStart,RegistrationEnd,Visibility )

TicketType=> Type: ( Id,EventId,TenantId,Name,Price,Capacity,ReservedQuantity,SalesStart,SalesEnd,RowVersion)

Ticket=> Type: ( Id,EventId,TenantId,AttendeeId,TicketTypeId,OrderId,TicketNumber,Status,QrCode,CheckedInAt )

Order=> Type: ( Id,EventId,TenantId,AttendeeId,OrderNumber,OrderStatus,CreatedAt,PaidAt,Total,OrderItems )

OrderItem=> Type: ( Id,TenantId,OrderId,TicketTypeId,Quantity,UnitPrice )

Payment=> Type: ( Id,OrderId,Amount,PaymentStatus,ProviderTransactionId)

*Notes* :
    - AttendeeId => user acts as Attendee 
    - ReservedQuantity (int) 

**Enums:

TicketStatus=> Enum : (CheckedIn,Cancelled,Valid)

EventStatus=> Enum : ( Draft - Published - RegistrationOpen - RegistrationClosed - Completed - Cancelled )

EventVisibility => Enum : ( Public,Private )

UserRole=> Enum : ( Manager,Employee )

OrderStatus=> Enum : ( Pending,Paid,Expired,Cancelled )

PaymentStatus=> Enum : ( Pending,Succeeded,Failed )
